cmake_minimum_required(VERSION 3.16)
project(iccad_d_starter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cadd0000 src/main.cpp src/parser.cpp src/analysis.cpp src/writer.cpp src/scoring.cpp src/operations.cpp src/legality.cpp src/optimizer.cpp)
target_include_directories(cadd0000 PRIVATE include)
target_compile_options(cadd0000 PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(test_operations tests/test_operations.cpp src/parser.cpp src/analysis.cpp src/writer.cpp src/scoring.cpp src/operations.cpp src/legality.cpp src/optimizer.cpp)
target_include_directories(test_operations PRIVATE include)
add_test(NAME operations COMMAND test_operations ${CMAKE_CURRENT_SOURCE_DIR}/examples/pdf_demo)
